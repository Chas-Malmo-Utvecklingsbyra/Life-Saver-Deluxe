91static inline bool rightAngle(const Matrix& m)
92{
93 auto radian = fabsf(tvg::atan2(m.e21, m.e11));

192static inline float length(const Point* a, const Point* b)
193{
194 auto x = b->x - a->x;
195 auto y = b->y - a->y;
196
197 if (x < 0) x = -x;
198 if (y < 0) y = -y;
199
200 return (x > y) ? (x + 0.375f * y) : (y + 0.375f * x);
201}
