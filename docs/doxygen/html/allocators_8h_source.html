85class CrtAllocator {
86public:
87 static const bool kNeedFree = true;
88 void* Malloc(size_t size) {
89 if (size) { // behavior of malloc(0) is implementation defined.
90 void * p = RAPIDJSON_MALLOC(size);
91 LV_ASSERT_MALLOC(p);
92 return p;
93 } else {
94 return NULL; // standardize to returning NULL.
95 }
96 }
97 void* Realloc(void* originalPtr, size_t originalSize, size_t newSize) {
98 (void)originalSize;
99 if (newSize == 0) {
100 RAPIDJSON_FREE(originalPtr);
101 return NULL;
102 }
103 void * p = RAPIDJSON_REALLOC(originalPtr, newSize);
104 LV_ASSERT_MALLOC(p);
105 return p;
106 }
107 static void Free(void *ptr) RAPIDJSON_NOEXCEPT { RAPIDJSON_FREE(ptr); }
108
109 bool operator==(const CrtAllocator&) const RAPIDJSON_NOEXCEPT {
110 return true;
111 }
112 bool operator!=(const CrtAllocator&) const RAPIDJSON_NOEXCEPT {
113 return false;
114 }
115};

136template <typename BaseAllocator = CrtAllocator>
137class MemoryPoolAllocator {
139
141 struct ChunkHeader {
142 size_t capacity;
143 size_t size;
144 ChunkHeader *next;
145 };
146
147 struct SharedData {
148 ChunkHeader *chunkHead;
149 BaseAllocator* ownBaseAllocator;
150 size_t refcount;
151 bool ownBuffer;
152 };
153
154 static const size_t SIZEOF_SHARED_DATA = RAPIDJSON_ALIGN(sizeof(SharedData));
155 static const size_t SIZEOF_CHUNK_HEADER = RAPIDJSON_ALIGN(sizeof(ChunkHeader));
156
157 static inline ChunkHeader *GetChunkHead(SharedData *shared)
158 {
159 return reinterpret_cast<ChunkHeader*>(reinterpret_cast<uint8_t*>(shared) + SIZEOF_SHARED_DATA);
160 }
161 static inline uint8_t *GetChunkBuffer(SharedData *shared)
162 {
163 return reinterpret_cast<uint8_t*>(shared->chunkHead) + SIZEOF_CHUNK_HEADER;
164 }
165
166 static const size_t kDefaultChunkCapacity = RAPIDJSON_ALLOCATOR_DEFAULT_CHUNK_CAPACITY;
167
168public:
169 static const bool kNeedFree = false;
170 static const bool kRefCounted = true;
171
173
176 explicit
177 MemoryPoolAllocator(size_t chunkSize = kDefaultChunkCapacity, BaseAllocator* baseAllocator = 0) :
178 chunk_capacity_(chunkSize),
179 baseAllocator_(baseAllocator ? baseAllocator : RAPIDJSON_NEW(BaseAllocator)()),
180 shared_(static_cast<SharedData*>(baseAllocator_ ? baseAllocator_->Malloc(SIZEOF_SHARED_DATA + SIZEOF_CHUNK_HEADER) : 0))
181 {
182 RAPIDJSON_ASSERT(baseAllocator_ != 0);
183 RAPIDJSON_ASSERT(shared_ != 0);
184 if (baseAllocator) {
185 shared_->ownBaseAllocator = 0;
186 }
187 else {
188 shared_->ownBaseAllocator = baseAllocator_;
189 }
190 shared_->chunkHead = GetChunkHead(shared_);
191 shared_->chunkHead->capacity = 0;
192 shared_->chunkHead->size = 0;
193 shared_->chunkHead->next = 0;
194 shared_->ownBuffer = true;
195 shared_->refcount = 1;
196 }
197
199
208 MemoryPoolAllocator(void *buffer, size_t size, size_t chunkSize = kDefaultChunkCapacity, BaseAllocator* baseAllocator = 0) :
209 chunk_capacity_(chunkSize),
210 baseAllocator_(baseAllocator),
211 shared_(static_cast<SharedData*>(AlignBuffer(buffer, size)))
212 {
213 RAPIDJSON_ASSERT(size >= SIZEOF_SHARED_DATA + SIZEOF_CHUNK_HEADER);
214 shared_->chunkHead = GetChunkHead(shared_);
215 shared_->chunkHead->capacity = size - SIZEOF_SHARED_DATA - SIZEOF_CHUNK_HEADER;
216 shared_->chunkHead->size = 0;
217 shared_->chunkHead->next = 0;
218 shared_->ownBaseAllocator = 0;
219 shared_->ownBuffer = false;
220 shared_->refcount = 1;
221 }
222
223 MemoryPoolAllocator(const MemoryPoolAllocator& rhs) RAPIDJSON_NOEXCEPT :
224 chunk_capacity_(rhs.chunk_capacity_),
225 baseAllocator_(rhs.baseAllocator_),
226 shared_(rhs.shared_)
227 {
228 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
229 ++shared_->refcount;
230 }
231 MemoryPoolAllocator& operator=(const MemoryPoolAllocator& rhs) RAPIDJSON_NOEXCEPT
232 {
233 RAPIDJSON_NOEXCEPT_ASSERT(rhs.shared_->refcount > 0);
234 ++rhs.shared_->refcount;
235 this->~MemoryPoolAllocator();
236 baseAllocator_ = rhs.baseAllocator_;
237 chunk_capacity_ = rhs.chunk_capacity_;
238 shared_ = rhs.shared_;
239 return *this;
240 }
241
242#if RAPIDJSON_HAS_CXX11_RVALUE_REFS
243 MemoryPoolAllocator(MemoryPoolAllocator&& rhs) RAPIDJSON_NOEXCEPT :
244 chunk_capacity_(rhs.chunk_capacity_),
245 baseAllocator_(rhs.baseAllocator_),
246 shared_(rhs.shared_)
247 {
248 RAPIDJSON_NOEXCEPT_ASSERT(rhs.shared_->refcount > 0);
249 rhs.shared_ = 0;
250 }
251 MemoryPoolAllocator& operator=(MemoryPoolAllocator&& rhs) RAPIDJSON_NOEXCEPT
252 {
253 RAPIDJSON_NOEXCEPT_ASSERT(rhs.shared_->refcount > 0);
254 this->~MemoryPoolAllocator();
255 baseAllocator_ = rhs.baseAllocator_;
256 chunk_capacity_ = rhs.chunk_capacity_;
257 shared_ = rhs.shared_;
258 rhs.shared_ = 0;
259 return *this;
260 }
261#endif
262
264
266 ~MemoryPoolAllocator() RAPIDJSON_NOEXCEPT {
267 if (!shared_) {
268 // do nothing if moved
269 return;
270 }
271 if (shared_->refcount > 1) {
272 --shared_->refcount;
273 return;
274 }
275 Clear();
276 BaseAllocator *a = shared_->ownBaseAllocator;
277 if (shared_->ownBuffer) {
278 baseAllocator_->Free(shared_);
279 }
280 RAPIDJSON_DELETE(a);
281 }
282
284 void Clear() RAPIDJSON_NOEXCEPT {
285 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
286 for (;;) {
287 ChunkHeader* c = shared_->chunkHead;
288 if (!c->next) {
289 break;
290 }
291 shared_->chunkHead = c->next;
292 baseAllocator_->Free(c);
293 }
294 shared_->chunkHead->size = 0;
295 }
296
298
300 size_t Capacity() const RAPIDJSON_NOEXCEPT {
301 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
302 size_t capacity = 0;
303 for (ChunkHeader* c = shared_->chunkHead; c != 0; c = c->next)
304 capacity += c->capacity;
305 return capacity;
306 }
307
309
311 size_t Size() const RAPIDJSON_NOEXCEPT {
312 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
313 size_t size = 0;
314 for (ChunkHeader* c = shared_->chunkHead; c != 0; c = c->next)
315 size += c->size;
316 return size;
317 }
318
320
322 bool Shared() const RAPIDJSON_NOEXCEPT {
323 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
324 return shared_->refcount > 1;
325 }
326
328 void* Malloc(size_t size) {
329 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
330 if (!size)
331 return NULL;
332
333 size = RAPIDJSON_ALIGN(size);
334 if (RAPIDJSON_UNLIKELY(shared_->chunkHead->size + size > shared_->chunkHead->capacity))
335 if (!AddChunk(chunk_capacity_ > size ? chunk_capacity_ : size))
336 return NULL;
337
338 void *buffer = GetChunkBuffer(shared_) + shared_->chunkHead->size;
339 shared_->chunkHead->size += size;
340 return buffer;
341 }
342
344 void* Realloc(void* originalPtr, size_t originalSize, size_t newSize) {
345 if (originalPtr == 0)
346 return Malloc(newSize);
347
348 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
349 if (newSize == 0)
350 return NULL;
351
352 originalSize = RAPIDJSON_ALIGN(originalSize);
353 newSize = RAPIDJSON_ALIGN(newSize);
354
355 // Do not shrink if new size is smaller than original
356 if (originalSize >= newSize)
357 return originalPtr;
358
359 // Simply expand it if it is the last allocation and there is sufficient space
360 if (originalPtr == GetChunkBuffer(shared_) + shared_->chunkHead->size - originalSize) {
361 size_t increment = static_cast<size_t>(newSize - originalSize);
362 if (shared_->chunkHead->size + increment <= shared_->chunkHead->capacity) {
363 shared_->chunkHead->size += increment;
364 return originalPtr;
365 }
366 }
367
368 // Realloc process: allocate and copy memory, do not free original buffer.
369 if (void* newBuffer = Malloc(newSize)) {
370 if (originalSize)
371 std::memcpy(newBuffer, originalPtr, originalSize);
372 return newBuffer;
373 }
374 else
375 return NULL;
376 }
377
379 static void Free(void *ptr) RAPIDJSON_NOEXCEPT { (void)ptr; } // Do nothing
380
382 bool operator==(const MemoryPoolAllocator& rhs) const RAPIDJSON_NOEXCEPT {
383 RAPIDJSON_NOEXCEPT_ASSERT(shared_->refcount > 0);
384 RAPIDJSON_NOEXCEPT_ASSERT(rhs.shared_->refcount > 0);
385 return shared_ == rhs.shared_;
386 }
387
388 bool operator!=(const MemoryPoolAllocator& rhs) const RAPIDJSON_NOEXCEPT {
389 return !operator==(rhs);
390 }
391
392private:
394
397 bool AddChunk(size_t capacity) {
398 if (!baseAllocator_)
399 shared_->ownBaseAllocator = baseAllocator_ = RAPIDJSON_NEW(BaseAllocator)();
400 if (ChunkHeader* chunk = static_cast<ChunkHeader*>(baseAllocator_->Malloc(SIZEOF_CHUNK_HEADER + capacity))) {
401 chunk->capacity = capacity;
402 chunk->size = 0;
403 chunk->next = shared_->chunkHead;
404 shared_->chunkHead = chunk;
405 return true;
406 }
407 else
408 return false;
409 }
410
411 static inline void* AlignBuffer(void* buf, size_t &size)
412 {
413 RAPIDJSON_NOEXCEPT_ASSERT(buf != 0);
414 const uintptr_t mask = sizeof(void*) - 1;
415 const uintptr_t ubuf = reinterpret_cast<uintptr_t>(buf);
416 if (RAPIDJSON_UNLIKELY(ubuf & mask)) {
417 const uintptr_t abuf = (ubuf + mask) & ~mask;
418 RAPIDJSON_ASSERT(size >= abuf - ubuf);
419 buf = reinterpret_cast<void*>(abuf);
420 size -= abuf - ubuf;
421 }
422 return buf;
423 }
424
425 size_t chunk_capacity_;
426 BaseAllocator* baseAllocator_;
427 SharedData *shared_;
428};

441template<typename T, typename A>
442inline T* Realloc(A& a, T* old_p, size_t old_n, size_t new_n)
443{
444 RAPIDJSON_NOEXCEPT_ASSERT(old_n <= (std::numeric_limits<size_t>::max)() / sizeof(T) && new_n <= (std::numeric_limits<size_t>::max)() / sizeof(T));
445 return static_cast<T*>(a.Realloc(old_p, old_n * sizeof(T), new_n * sizeof(T)));
446}

448template<typename T, typename A>
449inline T *Malloc(A& a, size_t n = 1)
450{
451 return Realloc<T, A>(a, NULL, 0, n);
452}

454template<typename T, typename A>
455inline void Free(A& a, T *p, size_t n = 1)
456{
457 static_cast<void>(Realloc<T, A>(a, p, n, 0));
458}
