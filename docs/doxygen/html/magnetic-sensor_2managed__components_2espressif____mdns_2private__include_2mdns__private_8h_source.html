115#define queueToEnd(type, queue, item) \
116 if (!queue) { \
117 queue = item; \
118 } else { \
119 type * _q = queue; \
120 while (_q->next) { _q = _q->next; } \
121 _q->next = item; \
122 }
123
124#define queueDetach(type, queue, item) \
125 if (queue) { \
126 if (queue == item) { \
127 queue = queue->next; \
128 } else { \
129 type * _q = queue; \
130 while (_q->next && _q->next != item) { \
131 _q = _q->next; \
132 } \
133 if (_q->next == item) { \
134 _q->next = item->next; \
135 item->next = NULL; \
136 } \
137 } \
138 }
139
140#define queueFree(type, queue) while (queue) { type * _q = queue; queue = queue->next; mdns_mem_free(_q); }
