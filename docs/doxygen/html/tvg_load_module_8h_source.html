33struct LoadModule
34{
35 INLIST_ITEM(LoadModule);
36
37 //Use either hashkey(data) or hashpath(path)
38 union {
39 uintptr_t hashkey;
40 char* hashpath = nullptr;
41 };
42
43 FileType type; //current loader file type
44 uint16_t sharing = 0; //reference count
45 bool readied = false; //read done already.
46 bool pathcache = false; //cached by path
47
48 LoadModule(FileType type) : type(type) {}
49 virtual ~LoadModule()
50 {
51 if (pathcache) lv_free(hashpath);
52 }
53
54 virtual bool open(const string& path) { return false; }
55 virtual bool open(const char* data, uint32_t size, bool copy) { return false; }
56 virtual bool resize(Paint* paint, float w, float h) { return false; }
57 virtual void sync() {}; //finish immediately if any async update jobs.
58
59 virtual bool read()
60 {
61 if (readied) return false;
62 readied = true;
63 return true;
64 }
65
66 bool cached()
67 {
68 if (hashkey) return true;
69 return false;
70 }
71
72 virtual bool close()
73 {
74 if (sharing == 0) return true;
75 --sharing;
76 return false;
77 }
78};
