48/* Expat tries very hard to make the API boundary very specifically
49 defined. There are two macros defined to control this boundary;
50 each of these can be defined before including this header to
51 achieve some different behavior, but doing so it not recommended or
52 tested frequently.
53
54 XMLCALL - The calling convention to use for all calls across the
55 "library boundary." This will default to cdecl, and
56 try really hard to tell the compiler that's what we
57 want.
58
59 XMLIMPORT - Whatever magic is needed to note that a function is
60 to be imported from a dynamically loaded library
61 (.dll, .so, or .sl, depending on your platform).
62
63 The XMLCALL macro was added in Expat 1.95.7. The only one which is
64 expected to be directly useful in client code is XMLCALL.
65
66 Note that on at least some Unix versions, the Expat library must be
67 compiled with the cdecl calling convention as the default since
68 system headers may assume the cdecl convention.
69*/
70#ifndef XMLCALL
71# if defined(_MSC_VER)
72# define XMLCALL __cdecl
73# elif defined(__GNUC__) && defined(__i386) && ! defined(__INTEL_COMPILER)
74# define XMLCALL __attribute__((cdecl))
75# else
76/* For any platform which uses this definition and supports more than
77 one calling convention, we need to extend this definition to
78 declare the convention used on that platform, if it's possible to
79 do so.
80
81 If this is the case for your platform, please file a bug report
82 with information on how to identify your platform via the C
83 pre-processor and how to specify the same calling convention as the
84 platform's malloc() implementation.
85*/
86# define XMLCALL
87# endif
88#endif /* not defined XMLCALL */
89
